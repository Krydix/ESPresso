cmake_minimum_required(VERSION 3.16)

set(ESPRESSO_BASE_VERSION "0.1.0")
if(DEFINED ENV{GITHUB_SHA} AND NOT "$ENV{GITHUB_SHA}" STREQUAL "")
    string(SUBSTRING "$ENV{GITHUB_SHA}" 0 7 ESPRESSO_GIT_VERSION)
else()
    execute_process(
        COMMAND git rev-parse --short=7 HEAD
        WORKING_DIRECTORY "${CMAKE_CURRENT_LIST_DIR}"
        OUTPUT_VARIABLE ESPRESSO_GIT_VERSION
        OUTPUT_STRIP_TRAILING_WHITESPACE
        ERROR_QUIET
    )
endif()
string(LENGTH "${ESPRESSO_GIT_VERSION}" ESPRESSO_GIT_VERSION_LENGTH)
if(ESPRESSO_GIT_VERSION_LENGTH EQUAL 7 AND
   ESPRESSO_GIT_VERSION MATCHES "^[0-9a-fA-F]+$")
    set(PROJECT_VER "${ESPRESSO_BASE_VERSION}-${ESPRESSO_GIT_VERSION}")
else()
    set(PROJECT_VER "${ESPRESSO_BASE_VERSION}")
endif()
set(PROJECT_VER_NUMBER 1)

include($ENV{IDF_PATH}/tools/cmake/project.cmake)
idf_build_set_property(DEPENDENCIES_LOCK dependencies.lock.${IDF_TARGET})
project(espresso)
